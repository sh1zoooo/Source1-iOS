cmake_minimum_required(VERSION 3.24)
project(Source1IOS LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)

include(cmake/SourceModules.cmake)
add_library(port_runtime STATIC src/runtime/Runtime.cpp src/source/SourceBridge.cpp src/source/SourceFiles.cpp src/source/SourceAppSystems.cpp src/source/SourceEngine.cpp src/source/SourceAssets.cpp src/source/SourceHost.cpp src/source/SourceMap.cpp src/source/SourceStudio.cpp src/source/SourcePlayer.cpp src/source/SourceMobile.cpp)
target_include_directories(port_runtime PUBLIC src/runtime)
target_include_directories(port_runtime PUBLIC src/source)
target_include_directories(port_runtime PRIVATE "${SOURCE_ROOT}/engine")
target_link_libraries(port_runtime PRIVATE source_modules)
if(SOURCE_LINK_CSTRIKE)
  target_compile_definitions(port_runtime PUBLIC SOURCE_GAME_LINK=1)
  add_library(source_player_sdk OBJECT src/source/SourcePlayerSDK.cpp)
  get_target_property(player_sdk_defines source_cstrike_server COMPILE_DEFINITIONS)
  get_target_property(player_sdk_includes source_cstrike_server INCLUDE_DIRECTORIES)
  target_compile_definitions(source_player_sdk PRIVATE ${player_sdk_defines})
  target_include_directories(source_player_sdk PRIVATE ${player_sdk_includes} src/source)
  target_link_libraries(source_player_sdk PRIVATE source_settings)
  set_target_properties(source_player_sdk PROPERTIES CXX_STANDARD 14)
  target_compile_options(source_player_sdk PRIVATE -w)
  target_sources(port_runtime PRIVATE $<TARGET_OBJECTS:source_player_sdk>)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  enable_language(OBJCXX)
  add_executable(Source1IOS MACOSX_BUNDLE src/ios/main.mm)
  set(SOURCE_NOTICES third_party/SOURCE1_LICENSE.txt third_party/SOURCE1_THIRD_PARTY_NOTICES.txt)
  set_source_files_properties(${SOURCE_NOTICES} PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  target_sources(Source1IOS PRIVATE ${SOURCE_NOTICES})
  target_link_libraries(Source1IOS PRIVATE port_runtime
    "-framework UIKit" "-framework Foundation" "-framework CoreGraphics"
    "-framework QuartzCore" "-framework Metal" "-framework MetalKit")
  target_compile_options(Source1IOS PRIVATE -fobjc-arc)
  set_target_properties(Source1IOS PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/src/ios/Info.plist"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "io.github.sh1zoooo.source1ios"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2"
    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO"
    XCODE_ATTRIBUTE_CODE_SIGNING_REQUIRED "NO"
    XCODE_ATTRIBUTE_ENABLE_BITCODE "NO"
    XCODE_ATTRIBUTE_CLANG_ENABLE_OBJC_ARC "YES"
    XCODE_ATTRIBUTE_GENERATE_INFOPLIST_FILE "NO")
else()
  include(CTest)
  if(BUILD_TESTING)
    add_executable(texture_atlas_tests tests/texture_atlas_tests.cpp)
    target_include_directories(texture_atlas_tests PRIVATE src/source)
    add_test(NAME texture_atlas_detail_bounds COMMAND texture_atlas_tests)
    if(SOURCE_LINK_CSTRIKE)
      add_executable(game_tests tests/game_tests.cpp)
      target_include_directories(game_tests PRIVATE "${SOURCE_ROOT}/engine")
      target_link_libraries(game_tests PRIVATE port_runtime source_modules)
      add_test(NAME cstrike_game_contract COMMAND game_tests)
      # Manual integration probe: requires user-supplied game content, never CI fixtures.
      add_executable(mobile_probe tests/mobile_probe.cpp)
      target_link_libraries(mobile_probe PRIVATE port_runtime source_modules)
      add_executable(world_probe tests/world_probe.cpp)
      target_link_libraries(world_probe PRIVATE port_runtime source_modules)
      add_executable(map_probe tests/map_probe.cpp)
      target_link_libraries(map_probe PRIVATE port_runtime source_modules)
      add_executable(gameplay_probe tests/gameplay_probe.cpp)
      target_link_libraries(gameplay_probe PRIVATE port_runtime source_modules)
      add_executable(clientmod_probe tests/clientmod_probe.cpp)
      target_include_directories(clientmod_probe PRIVATE "${SOURCE_ROOT}/engine")
      target_link_libraries(clientmod_probe PRIVATE port_runtime source_modules)
    endif()
    add_executable(friction_tests tests/friction_tests.cpp)
    target_link_libraries(friction_tests PRIVATE port_runtime source_modules)
    add_test(NAME friction_contact_lifetime COMMAND friction_tests)
    add_executable(runtime_tests tests/runtime_tests.cpp)
    target_link_libraries(runtime_tests PRIVATE port_runtime source_settings)
    add_test(NAME runtime_contract COMMAND runtime_tests)
    add_executable(studio_mutation tests/studio_mutation.cpp src/source/SourceStudio.cpp)
    target_include_directories(studio_mutation PRIVATE src/source)
    target_link_libraries(studio_mutation PRIVATE source_settings source_mathlib source_tier1 source_tier0)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_compile_options(studio_mutation PRIVATE -fsanitize=address -fno-omit-frame-pointer)
      target_link_options(studio_mutation PRIVATE -fsanitize=address)
      target_link_libraries(studio_mutation PRIVATE dl pthread)
    endif()
    add_executable(entities_mutation tests/entities_mutation.cpp)
    target_include_directories(entities_mutation PRIVATE src/source)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_compile_options(entities_mutation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(entities_mutation PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME entities_mutation_bounds COMMAND entities_mutation)
    add_executable(props_mutation tests/props_mutation.cpp)
    target_include_directories(props_mutation PRIVATE src/source)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_compile_options(props_mutation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(props_mutation PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME props_mutation_bounds COMMAND props_mutation)
    add_executable(phy_mutation tests/phy_mutation.cpp)
    target_include_directories(phy_mutation PRIVATE src/source)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
      target_compile_options(phy_mutation PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(phy_mutation PRIVATE -fsanitize=address,undefined)

    add_test(NAME phy_mutation_bounds COMMAND phy_mutation)
    set_tests_properties(phy_mutation_bounds PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0" TIMEOUT 60)
    set_tests_properties(props_mutation_bounds PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0" TIMEOUT 60)
    set_tests_properties(entities_mutation_bounds PROPERTIES ENVIRONMENT "ASAN_OPTIONS=detect_leaks=0" TIMEOUT 60)

  endif()
